To run the code using {\ttfamily  mpirun }, use the following command \+: ~\newline



\begin{DoxyCode}{0}
\DoxyCodeLine{\$\ mpirun\ -\/n\ process\_number\ code\_path\ -\/f\ input\_file\ }

\end{DoxyCode}



\begin{DoxyItemize}
\item process\+\_\+number \+: The number of processes to run. ~\newline

\item code\+\_\+path \+: The path to your executable code. ~\newline

\item input\+\_\+file \+: The input file required for the code to run. ~\newline
 (input file should be {\ttfamily .json} file) ~\newline

\end{DoxyItemize}

{\bfseries{Example}} 


\begin{DoxyCode}{0}
\DoxyCodeLine{\$\ mpirun\ -\/n\ 4\ /usr/local/bin/mycode\ -\/f\ data/input.json\ }

\end{DoxyCode}


{\bfseries{Possble}} {\bfseries{command-\/line}} {\bfseries{options}} 
\begin{DoxyItemize}
\item -\/h \+: Help option ~\newline

\item -\/f \+: \doxysectlink{InputFile_InputFile}{Input file}{1} option (Required) (See \doxysectlink{InputFile_InputFile}{Input file}{1}) ~\newline

\item -\/m \+: Method (See method) ~\newline

\item -\/q \+: Quantity (See quantity) ~\newline

\item -\/I \+: File path to set the position of bath spins (See bathfiles) ~\newline

\item -\/s \+: File path to set the spin state of bath spins (See statefile) ~\newline

\item -\/a \+: File path to set principal axies of the defects in the bath (See avaaxfile) ~\newline

\item -\/S \+: File path to set the spin state of on-\/site spins of the defects in the bath (See exstatefile) ~\newline

\item -\/N \+: The number of state configuration to do time ensemble average (See nstate) ~\newline

\item -\/B \+: Magnetic field in z direction (Unit \+: G) (See bfield) ~\newline

\item -\/o \+: Result file path (See output) ~\newline

\end{DoxyItemize}
